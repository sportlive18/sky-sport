async function startPlayer(ch){ $pc.style.display="flex"; shaka.polyfill.installAll(); if(!shaka.Player.isBrowserSupported()){ errMsg("Browser not supported."); return; } // ← create player WITH video, no double overlay var player = new shaka.Player(); await player.attach($vid); var ui = new shaka.ui.Overlay(player, $pc, $vid); ui.configure({ controlPanelElements:["play_pause","mute","volume","time_and_duration","spacer","quality","fullscreen"], seekBarColors:{base:"#a8a8a8",buffered:"#7f1d1d",played:"#ef4444"} }); setTimeout(function(){ injectControls($pc,$vid,player); },150); var cookie = ch.cookie || ''; var keyId = ch.key_id || ''; var key = ch.key || ''; var url = ch.stream_url || ''; if (!url) { errMsg("Missing stream URL"); return; } // ── Config ───────────────────────────────────────────── var cfg = { manifest:{ retryParameters:{maxAttempts:3,baseDelay:500,backoffFactor:2,timeout:20000} }, streaming:{ lowLatencyMode:false, bufferingGoal: PROF.buf, rebufferingGoal: PROF.reBuf, bufferBehind: PROF.behind, retryParameters:{maxAttempts:3,baseDelay:500,backoffFactor:2,timeout:20000} }, abr:{ enabled:true, defaultBandwidthEstimate:PROF.bwEst, restrictions: PROF.maxH < 1080 ? { maxHeight: PROF.maxH } : {} } }; if (keyId && key) { cfg.drm = { clearKeys: {} }; cfg.drm.clearKeys[keyId] = key; } player.configure(cfg); // ── Only set Cookie header if cookie exists (browser may ignore it, but try) ── if (cookie) { player.getNetworkingEngine().registerRequestFilter(function(type, req){ // NOTE: browsers block these on cross-origin — this only works if the // CDN returns Access-Control-Allow-Headers allowing them, which JioTV does NOT. try { req.headers["Cookie"] = cookie; req.headers["Referer"] = "https://www.jiotv.com/"; req.headers["Origin"] = "https://www.jiotv.com"; } catch(e) { /* browser stripped them */ } }); } // ── Load ─────────────────────────────────────────────── try { toast("Loading…","warn",0); await player.load(url); hideOverlay(); $vid.muted = false; $vid.play().catch(function(){ $vid.muted = true; $vid.play().catch(function(){ toast("Tap ▶ to start","ok",5000); }); }); toast("● Live","ok",3000); } catch(e) { console.error("Load failed:", e); toast("Load failed: " + shakaErrStr(e), "err", 0); } // ── Recovery ─────────────────────────────────────────── player.addEventListener("error", function(ev){ console.error("Player error:", ev.detail); toast(shakaErrStr(ev.detail), "err", 6000); }); $vid.addEventListener("waiting", function(){ toast("Buffering…","warn",2000); }); $vid.addEventListener("playing", function(){ toast("● Live","ok",2000); }); }
